cmake_minimum_required(VERSION 3.24)
project(NorthpassCore VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NORTHPASS_SANITIZERS "Address/undefined sanitizers for portable tests" OFF)
find_package(Threads REQUIRED)
add_library(northpass_packets STATIC src/packet.cpp src/flow.cpp src/options.cpp src/metrics.cpp src/ipc_protocol.cpp src/transformation.cpp)
target_link_libraries(northpass_packets PUBLIC Threads::Threads)
target_include_directories(northpass_packets PUBLIC include)
if(MSVC)
  set_property(TARGET northpass_packets PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(northpass_packets PRIVATE /W4 /WX /permissive- /sdl /guard:cf)
else()
  target_compile_options(northpass_packets PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(NORTHPASS_SANITIZERS)
    target_compile_options(northpass_packets PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(northpass_packets PUBLIC -fsanitize=address,undefined)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(NorthpassCoreTests tests/packet_tests.cpp)
  target_link_libraries(NorthpassCoreTests PRIVATE northpass_packets)
  if(MSVC)
    set_property(TARGET NorthpassCoreTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(NorthpassCoreTests PRIVATE /W4 /WX /permissive-)
  endif()
  foreach(case ipv4 ipv6 tcp udp tls fragments malformed tracking invariant options)
    add_test(NAME native_${case} COMMAND NorthpassCoreTests ${case})
    set_tests_properties(native_${case} PROPERTIES TIMEOUT 30)
  endforeach()
  add_executable(NorthpassReliabilityTests tests/reliability_tests.cpp)
  target_link_libraries(NorthpassReliabilityTests PRIVATE northpass_packets)
  if(MSVC)
    set_property(TARGET NorthpassReliabilityTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(NorthpassReliabilityTests PRIVATE /W4 /WX /permissive-)
  endif()
  foreach(case validation retransmissions fragment_details flow_expiry strategies recovery queue concurrency memory performance metrics ipc_codec)
    add_test(NAME native_${case} COMMAND NorthpassReliabilityTests ${case})
    set_tests_properties(native_${case} PROPERTIES TIMEOUT 30)
  endforeach()
  add_executable(NorthpassFoundationTests tests/foundation_tests.cpp)
  target_link_libraries(NorthpassFoundationTests PRIVATE northpass_packets)
  if(MSVC)
    set_property(TARGET NorthpassFoundationTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(NorthpassFoundationTests PRIVATE /W4 /WX /permissive-)
  endif()
  foreach(case transformation rollback lifecycle checksums mtu fault_scope benchmark)
    add_test(NAME native_${case} COMMAND NorthpassFoundationTests ${case})
    set_tests_properties(native_${case} PROPERTIES TIMEOUT 30)
  endforeach()
endif()
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
    message(FATAL_ERROR "NorthpassCore supports Visual Studio Windows x64 only (-A x64).")
  endif()
  if(NOT NORTHPASS_WINDIVERT_ROOT)
    message(FATAL_ERROR "Run scripts/prepare-native.py --dependency-only; supply NORTHPASS_WINDIVERT_ROOT.")
  endif()
  enable_language(RC)
  # CMake independently rechecks the SDK header and runtime bytes before compilation.
  foreach(entry "include/windivert.h|5017a1768c1592fd664c0c2d3d2d30f81fad4ab98d322b1914c6f0a33fcacdf9" "bin/WinDivert.dll|c1e060ee19444a259b2162f8af0f3fe8c4428a1c6f694dce20de194ac8d7d9a2" "bin/WinDivert64.sys|8da085332782708d8767bcace5327a6ec7283c17cfb85e40b03cd2323a90ddc2")
    string(REPLACE "|" ";" parts "${entry}")
    list(GET parts 0 path)
    list(GET parts 1 expected)
    file(SHA256 "${NORTHPASS_WINDIVERT_ROOT}/${path}" actual)
    if(NOT actual STREQUAL expected)
      message(FATAL_ERROR "Untrusted WinDivert input: ${path}")
    endif()
  endforeach()
  add_executable(NorthpassCore src/windows_main.cpp src/windows_security.cpp src/windows_ipc.cpp src/version.rc)
  target_include_directories(NorthpassCore PRIVATE "${NORTHPASS_WINDIVERT_ROOT}/include")
  target_compile_definitions(NorthpassCore PRIVATE WINVER=0x0A00 _WIN32_WINNT=0x0A00 NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
  target_link_libraries(NorthpassCore PRIVATE northpass_packets bcrypt advapi32 shell32 ole32 psapi)
  set_property(TARGET NorthpassCore PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(NorthpassCore PRIVATE /W4 /WX /permissive- /sdl /guard:cf /utf-8)
  target_link_options(NorthpassCore PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /CETCOMPAT /guard:cf)
  if(NORTHPASS_BROKER_HEADER)
    add_executable(NorthpassBrokerBootstrap src/windows_broker.cpp src/windows_security.cpp)
    target_include_directories(NorthpassBrokerBootstrap PRIVATE include "${NORTHPASS_BROKER_HEADER}")
    target_compile_definitions(NorthpassBrokerBootstrap PRIVATE WINVER=0x0A00 _WIN32_WINNT=0x0A00 NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
    target_link_libraries(NorthpassBrokerBootstrap PRIVATE bcrypt advapi32 shell32 ole32)
    set_property(TARGET NorthpassBrokerBootstrap PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(NorthpassBrokerBootstrap PRIVATE /W4 /WX /permissive- /sdl /guard:cf /utf-8)
    target_link_options(NorthpassBrokerBootstrap PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /CETCOMPAT /guard:cf /MANIFESTUAC:level='requireAdministrator')
  endif()
endif()
